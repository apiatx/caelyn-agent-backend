async def refresh_watchlist(watchlist_id: str):

    BATCH_SIZE = 22

                print(f"Batch {batch_index} failed: {e}")

    tasks = [analyze_batch(b, i) for i, b in enumerate(batches)]

@router.post("/api/watchlist")
async def create_watchlist(body: dict):
    try:
        name = body.get("name", "Untitled")
        tickers = body.get("tickers", [])    # list of strings
        csv_data = body.get("csv_data", [])  # list of row dicts
        watchlist_id = str(uuid.uuid4())[:8]
        watchlist = {
            "id": watchlist_id,
            "name": name,
            "tickers": tickers,
            "csv_data": csv_data,
            "analysis": None,
            "created_at": datetime.utcnow().isoformat() + "Z"
        }
        await save_watchlist(watchlist)
        return watchlist
    except Exception as e:
        import traceback
        traceback.print_exc()
        raise HTTPException(500, detail=str(e))  # return the REAL error

Quick Debugging Tip
Wrap every route in a bare try/except Exception as e and raise HTTPException(500, detail=str(e)) — the HTTP 000 you're seeing means FastAPI never gets a chance to serialize the error, so there's something crashing outside the normal exception path (probably in middleware, a startup event, or a dependency that's None).
